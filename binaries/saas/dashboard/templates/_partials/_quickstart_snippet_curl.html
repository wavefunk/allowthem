Works today against any allowthem tenant. 

# Discover the OIDC config
curl {{ issuer }}/.well-known/openid-configuration

# Exchange an authorization code for tokens (after running the user
# through /authorize and capturing the code on your callback)
curl -X POST {{ issuer }}/oauth/token \
  -u '{{ client_id }}:{{ client_secret }}' \
  -d 'grant_type=authorization_code' \
  -d 'code=<CODE_FROM_REDIRECT>' \
  -d 'redirect_uri={{ redirect_uri }}'  # TODO: replace with your real redirect URI
