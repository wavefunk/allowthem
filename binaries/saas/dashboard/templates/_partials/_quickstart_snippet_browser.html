// Browser — PKCE authorization code flow
import { createAllowthemClient } from '@allowthem/js';

const auth = createAllowthemClient({
  issuer: '{{ issuer }}',
  clientId: '{{ client_id }}',
  redirectUri: window.location.origin + '/callback',
});

await auth.signinRedirect();
